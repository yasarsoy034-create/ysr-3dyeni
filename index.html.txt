    <script>
        const WHATSAPP_NUMARASI = "905000000000"; // TELEFON NUMARANIZI BURAYA YAZIN

        let aktifUrun = "gardirop";
        let urunAdlari = {
            gardirop: "Gardırop / Elbise Dolabı",
            mutfak: "Mutfak Dolabı",
            vestiyer: "Vestiyer / Portmanto",
            tv: "TV Ünitesi & Lambiri Pano"
        };

        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x0b0f19);

        const camera = new THREE.PerspectiveCamera(45, container.clientWidth / container.clientHeight, 0.1, 1000);
        camera.position.set(220, 180, 300);

        const renderer = new THREE.WebGLRenderer({ antialias: true, preserveDrawingBuffer: true });
        renderer.setSize(container.clientWidth, container.clientHeight);
        renderer.shadowMap.enabled = true;
        container.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.target.set(0, 80, 0);

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.85);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xffffff, 0.9);
        dirLight.position.set(120, 200, 100);
        dirLight.castShadow = true;
        scene.add(dirLight);

        const grid = new THREE.GridHelper(500, 50, 0x374151, 0x1f2937);
        scene.add(grid);

        const malzemeler = {
            ahsap: new THREE.MeshStandardMaterial({ color: 0xc29b38, roughness: 0.4 }),
            gri: new THREE.MeshStandardMaterial({ color: 0x4b5563, roughness: 0.3 }),
            siyah: new THREE.MeshStandardMaterial({ color: 0x111827, roughness: 0.7 }),
            ic_govde: new THREE.MeshStandardMaterial({ color: 0x1f2937, roughness: 0.6 }),
            tezgah: new THREE.MeshStandardMaterial({ color: 0xe5e7eb, roughness: 0.2 })
        };

        let seciliMalzeme = malzemeler.ahsap;
        let seciliMalzemeAdi = "Meşe Ahşap";
        let kapakAcik = false;
        let modelGrubu = new THREE.Group();
        let solKapakPivot = new THREE.Group();
        let sagKapakPivot = new THREE.Group();

        scene.add(modelGrubu);

        function urunDegistir(urunKey) {
            aktifUrun = urunKey;
            ['gardirop', 'mutfak', 'vestiyer', 'tv'].forEach(key => {
                const btn = document.getElementById('btn-' + key);
                if (key === urunKey) {
                    btn.className = "p-2.5 text-xs rounded-lg border border-amber-500 bg-amber-500/20 text-amber-300 font-bold text-left transition";
                } else {
                    btn.className = "p-2.5 text-xs rounded-lg border border-gray-700 bg-gray-900 text-gray-400 font-bold text-left transition hover:border-gray-500";
                }
            });
            guncelle();
        }

        function renkDegistir(renkKey, renkEtiket) {
            seciliMalzeme = malzemeler[renkKey];
            seciliMalzemeAdi = renkEtiket;
            guncelle();
        }

        function kapakTetikle() {
            kapakAcik = !kapakAcik;
            const aci = kapakAcik ? Math.PI * 0.55 : 0;
            gsap.to(solKapakPivot.rotation, { y: -aci, duration: 1.2, ease: "power2.out" });
            gsap.to(sagKapakPivot.rotation, { y: aci, duration: 1.2, ease: "power2.out" });
        }

        function guncelle() {
            const g = parseFloat(document.getElementById('genislik').value);
            const y = parseFloat(document.getElementById('yukseklik').value);
            const d = parseFloat(document.getElementById('derinlik').value);

            document.getElementById('val-g').innerText = g + ' cm';
            document.getElementById('val-y').innerText = y + ' cm';
            document.getElementById('val-d').innerText = d + ' cm';

            const g_m = g / 100; const y_m = y / 100; const d_m = d / 100;
            let m2 = (2 * y_m * d_m) + (2 * g_m * d_m) + (g_m * y_m);
            let bant = (4 * y_m) + (8 * d_m) + (8 * g_m);
            let katsayi = 950;

            if (aktifUrun === 'mutfak') katsayi = 1250;
            if (aktifUrun === 'tv') katsayi = 850;

            const fiyat = (m2 * katsayi) + (bant * 25) + 2000;

            document.getElementById('fiyat').innerText = Math.round(fiyat).toLocaleString('tr-TR') + ' TL';
            document.getElementById('m2-alan').innerText = m2.toFixed(2) + ' m²';
            document.getElementById('bant-metraj').innerText = bant.toFixed(1) + ' m';

            // Kesim Listesi Güncelleme
            let kesimHTML = "";
            if (aktifUrun === 'gardirop') {
                kesimHTML = `
                    <div class="flex justify-between"><span>Yan Panel (2x):</span> <b class="text-gray-200">${d*10} x ${y*10} mm</b></div>
                    <div class="flex justify-between"><span>Alt/Üst Tabla (2x):</span> <b class="text-gray-200">${d*10} x ${(g-3.6)*10} mm</b></div>
                    <div class="flex justify-between"><span>Orta Raf (2x):</span> <b class="text-gray-200">${(d-2)*10} x ${(g-3.6)*10} mm</b></div>
                    <div class="flex justify-between"><span>Kapaklar (2x):</span> <b class="text-gray-200">${((g/2)-0.4)*10} x ${(y-0.6)*10} mm</b></div>
                `;
            } else if (aktifUrun === 'mutfak') {
                kesimHTML = `
                    <div class="flex justify-between"><span>Alt Modül Yan (2x):</span> <b class="text-gray-200">${d*10} x 720 mm</b></div>
                    <div class="flex justify-between"><span>Mutfak Tezgahı:</span> <b class="text-gray-200">${(d+5)*10} x ${g*10} mm</b></div>
                    <div class="flex justify-between"><span>Üst Modül Yan (2x):</span> <b class="text-gray-200">350 x 720 mm</b></div>
                    <div class="flex justify-between"><span>Kapaklar (4x):</span> <b class="text-gray-200">${((g/2)-0.4)*10} x 715 mm</b></div>
                `;
            } else if (aktifUrun === 'vestiyer') {
                kesimHTML = `
                    <div class="flex justify-between"><span>Gövde Dikme (2x):</span> <b class="text-gray-200">${d*10} x ${y*10} mm</b></div>
                    <div class="flex justify-between"><span>Oturma Minderi Baza:</span> <b class="text-gray-200">${d*10} x ${(g/2)*10} mm</b></div>
                    <div class="flex justify-between"><span>Ayakkabılık Kapak:</span> <b class="text-gray-200">${(g/2)*10} x 450 mm</b></div>
                    <div class="flex justify-between"><span>Ayna Arka Pano:</span> <b class="text-gray-200">${(g/2)*10} x ${(y-50)*10} mm</b></div>
                `;
            } else if (aktifUrun === 'tv') {
                kesimHTML = `
                    <div class="flex justify-between"><span>Lambiri Pano Arkalık:</span> <b class="text-gray-200">${g*10} x ${y*10} mm</b></div>
                    <div class="flex justify-between"><span>TV Konsolu Üst Tabla:</span> <b class="text-gray-200">${d*10} x ${g*10} mm</b></div>
                    <div class="flex justify-between"><span>Çekmece Klapaları (2x):</span> <b class="text-gray-200">${(g/2)*10} x 300 mm</b></div>
                `;
            }
            document.getElementById('kesim-listesi').innerHTML = kesimHTML;

            // 3D Çizimi Yenile
            scene.remove(modelGrubu);
            modelGrubu = new THREE.Group();
            const t = 1.8;

            if (aktifUrun === 'gardirop') {
                const govdeGeo = new THREE.BoxGeometry(g, y, d);
                const govdeMesh = new THREE.Mesh(govdeGeo, seciliMalzeme);
                govdeMesh.position.y = y / 2;
                modelGrubu.add(govdeMesh);

                const icGeo = new THREE.BoxGeometry(g - (t*2), y - (t*2), d - t);
                const icMesh = new THREE.Mesh(icGeo, malzemeler.ic_govde);
                icMesh.position.set(0, y / 2, t / 2);
                modelGrubu.add(icMesh);

                solKapakPivot = new THREE.Group();
                solKapakPivot.position.set(-g / 2, y / 2, d / 2);
                const kapakGenislik = (g / 2) - 0.2;
                const solKapakGeo = new THREE.BoxGeometry(kapakGenislik, y - 0.4, t);
                const solKapakMesh = new THREE.Mesh(solKapakGeo, seciliMalzeme);
                solKapakMesh.position.set(kapakGenislik / 2, 0, 0);
                solKapakPivot.add(solKapakMesh);
                modelGrubu.add(solKapakPivot);

                sagKapakPivot = new THREE.Group();
                sagKapakPivot.position.set(g / 2, y / 2, d / 2);
                const sagKapakGeo = new THREE.BoxGeometry(kapakGenislik, y - 0.4, t);
                const sagKapakMesh = new THREE.Mesh(sagKapakGeo, seciliMalzeme);
                sagKapakMesh.position.set(-kapakGenislik / 2, 0, 0);
                sagKapakPivot.add(sagKapakPivot);
                modelGrubu.add(sagKapakPivot);

            } else if (aktifUrun === 'mutfak') {
                // Alt Dolap
                const altGeo = new THREE.BoxGeometry(g, 72, d);
                const altMesh = new THREE.Mesh(altGeo, seciliMalzeme);
                altMesh.position.y = 36;
                modelGrubu.add(altMesh);

                // Tezgah
                const tezgahGeo = new THREE.BoxGeometry(g + 4, 4, d + 4);
                const tezgahMesh = new THREE.Mesh(tezgahGeo, malzemeler.tezgah);
                tezgahMesh.position.y = 74;
                modelGrubu.add(tezgahMesh);

                // Üst Dolap
                const ustGeo = new THREE.BoxGeometry(g, 72, 35);
                const ustMesh = new THREE.Mesh(ustGeo, seciliMalzeme);
                ustMesh.position.set(0, 180, (d/2) - 17.5);
                modelGrubu.add(ustMesh);

            } else if (aktifUrun === 'tv') {
                // Arka Lambiri Pano
                const panoGeo = new THREE.BoxGeometry(g, y, 4);
                const panoMesh = new THREE.Mesh(panoGeo, seciliMalzeme);
                panoMesh.position.y = y / 2;
                modelGrubu.add(panoMesh);

                // TV Konsolu
                const konsolGeo = new THREE.BoxGeometry(g * 0.8, 35, d);
                const konsolMesh = new THREE.Mesh(konsolGeo, malzemeler.ic_govde);
                konsolMesh.position.set(0, 25, d / 2);
                modelGrubu.add(konsolMesh);

            } else if (aktifUrun === 'vestiyer') {
                // Vestiyer Gövde
                const vGeo = new THREE.BoxGeometry(g, y, d);
                const vMesh = new THREE.Mesh(vGeo, seciliMalzeme);
                vMesh.position.y = y / 2;
                modelGrubu.add(vMesh);
            }

            scene.add(modelGrubu);
        }

        function whatsappGonder() {
            const g = document.getElementById('genislik').value;
            const y = document.getElementById('yukseklik').value;
            const d = document.getElementById('derinlik').value;
            const fiyat = document.getElementById('fiyat').innerText;
            const urunEtiket = urunAdlari[aktifUrun];

            const mesaj = `🛠️ *YSR DESIGN - 3D Sipariş / Teklif Talebi*\n\n` +
                          `📦 *Seçilen Ürün:* ${urunEtiket}\n` +
                          `📏 *Ölçüler:*\n` +
                          `• Genişlik: ${g} cm\n` +
                          `• Yükseklik: ${y} cm\n` +
                          `• Derinlik: ${d} cm\n\n` +
                          `🎨 *Malzeme:* ${seciliMalzemeAdi}\n` +
                          `💰 *Hesaplanan Fiyat:* ${fiyat}\n\n` +
                          `📐 *İmalat Kesim Listesi (mm):*\n` +
                          document.getElementById('kesim-listesi').innerText.replace(/\n/g, '\n• ') + `\n\n` +
                          `Sipariş detaylarını onaylamak ve imalat sürecini başlatmak istiyorum.`;

            const url = `https://wa.me/${WHATSAPP_NUMARASI}?text=${encodeURIComponent(mesaj)}`;
            window.open(url, '_blank');
        }

        function arBaslat() {
            alert("AR Modu Başlatılıyor... Mobil cihazınızın kamerasını kullanarak ürünü odanıza yerleştirebilirsiniz.");
        }

        document.getElementById('genislik').addEventListener('input', guncelle);
        document.getElementById('yukseklik').addEventListener('input', guncelle);
        document.getElementById('derinlik').addEventListener('input', guncelle);

        window.addEventListener('resize', () => {
            camera.aspect = container.clientWidth / container.clientHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(container.clientWidth, container.clientHeight);
        });

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }

        guncelle();
        animate();
    </script>
